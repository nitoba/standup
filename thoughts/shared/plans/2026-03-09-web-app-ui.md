# Web App UI Implementation Plan

**Goal:** Implement the 4 desktop Angular pages (Login, Dashboard, Standup Detail, Settings) with the terminal-themed UI from `apps/web/design.pen`, using Tailwind v4 and signal-based mock data.

**Architecture:** A single shared sidebar layout component wraps the three authenticated pages, while Login remains full-screen. Data flows through a mock `StandupService` built on signals; pages read computed signals and update local state via service methods. Routing uses lazy-loaded `loadComponent` with `withComponentInputBinding()` to bind `:id` params for detail views.

**Design:** `thoughts/shared/designs/2026-03-09-web-app-ui-design.md`

```ts
import { Injectable, computed, signal } from '@angular/core'

import type {
  DashboardMetrics,
  Standup,
  StandupSection,
  StandupSourceRepo,
  StandupStatus,
} from '../types/standup'

const METRIC_CHANGES = {
  total: '++ 12 this_week',
  approved: '++ 8 this_week',
  pending: '++ 3 today',
  rejected: '++ 1 today',
} as const

const TARGET_TOTAL = 142
const TARGET_APPROVED = 128
const TARGET_PENDING = 8
const TARGET_REJECTED = 6

@Injectable({ providedIn: 'root' })
export class StandupService {
  private readonly standupsSignal = signal<Standup[]>(buildMockStandups())

  readonly standups = this.standupsSignal.asReadonly()

  readonly metrics = computed<DashboardMetrics>(() => {
    const standups = this.standups()
    const counts = standups.reduce(
      (acc, standup) => {
        acc.total += 1
        if (standup.status === 'approved') acc.approved += 1
        if (standup.status === 'pending_review') acc.pending += 1
        if (standup.status === 'rejected') acc.rejected += 1
        return acc
      },
      { total: 0, approved: 0, pending: 0, rejected: 0 },
    )

    return {
      total: { count: counts.total, change: METRIC_CHANGES.total },
      approved: { count: counts.approved, change: METRIC_CHANGES.approved },
      pending: { count: counts.pending, change: METRIC_CHANGES.pending },
      rejected: { count: counts.rejected, change: METRIC_CHANGES.rejected },
    }
  })

  getStandupById(id: string) {
    return computed(() => this.standups().find((standup) => standup.id === id))
  }

  approve(id: string) {
    this.updateStatus(id, 'approved')
  }

  reject(id: string) {
    this.updateStatus(id, 'rejected')
  }

  regenerate(id: string) {
    this.updateStatus(id, 'pending_review')
  }

  private updateStatus(id: string, status: StandupStatus) {
    this.standupsSignal.update((items) =>
      items.map((item) => (item.id === id ? { ...item, status } : item)),
    )
  }
}

function buildMockStandups(): Standup[] {
  const featured = buildFeaturedStandups()
  const remaining = TARGET_TOTAL - featured.length
  const statusPool = buildStatusPool(remaining)

  const filler = Array.from({ length: remaining }, (_, index) => {
    const status = statusPool[index] ?? 'approved'
    const day = String(6 - (index % 5)).padStart(2, '0')
    return {
      id: `auto-${index + 1}`,
      date: `2026-03-${day}`,
      status,
      createdAt: '17:20',
      contentPreview:
        'updated automation flows and reviewed standup generation outputs...',
      sections: buildDefaultSections(),
      sources: buildDefaultSources(),
    }
  })

  return [...featured, ...filler]
}

function buildStatusPool(remaining: number): StandupStatus[] {
  const approvedNeeded = TARGET_APPROVED - 3
  const pendingNeeded = TARGET_PENDING - 1
  const rejectedNeeded = TARGET_REJECTED - 1

  const pool: StandupStatus[] = [
    ...Array.from({ length: approvedNeeded }, () => 'approved'),
    ...Array.from({ length: pendingNeeded }, () => 'pending_review'),
    ...Array.from({ length: rejectedNeeded }, () => 'rejected'),
  ]

  return pool.slice(0, remaining)
}
```
